cmake_minimum_required(VERSION 3.24)
project(Arbitrium VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT MAIDIONIS_ROOT OR NOT ARBITRIUM_SOURCE_DIGEST)
  message(FATAL_ERROR "Use tools/configure.py with the pinned Maidionis source checkout")
endif()
add_subdirectory("${MAIDIONIS_ROOT}" "${CMAKE_BINARY_DIR}/maidionis")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
get_target_property(core_definitions maidionis_contracts INTERFACE_COMPILE_DEFINITIONS)
string(REGEX REPLACE "MAIDIONIS_BUILD_DIGEST=\"([^\"]+)\"" "\\1" MAIDIONIS_BUILD_DIGEST "${core_definitions}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependency.lock.json" lock)
string(JSON pinned_core GET "${lock}" core_build_digest)
if(NOT MAIDIONIS_BUILD_DIGEST STREQUAL pinned_core)
  message(FATAL_ERROR "Core build digest mismatch")
endif()
file(GLOB_RECURSE ARBITRIUM_IDENTITY_SOURCES CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"
  "include/*.h" "src/*.cpp" "python/arbitrium/*.py" "tools/*.py")
list(APPEND ARBITRIUM_IDENTITY_SOURCES "CMakeLists.txt" "dependency.lock.json")
list(SORT ARBITRIUM_IDENTITY_SOURCES)
set(identity_bytes "arbitrium-code-build-v1\n")
foreach(path IN LISTS ARBITRIUM_IDENTITY_SOURCES)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${path}")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${path}" h)
  string(APPEND identity_bytes "${path}\n${h}\n")
endforeach()
string(SHA256 actual_source_digest "${identity_bytes}")
if(NOT ARBITRIUM_SOURCE_DIGEST STREQUAL actual_source_digest)
  message(FATAL_ERROR "Source content changed; rerun tools/configure.py")
endif()
string(SHA256 ARBITRIUM_BUILD_DIGEST "arbitrium-composition-v1\n${MAIDIONIS_BUILD_DIGEST}\n${ARBITRIUM_SOURCE_DIGEST}\n")
add_library(arbitrium_decision STATIC src/composition.cpp)
target_link_libraries(arbitrium_decision PUBLIC maidionis_core maidionis_statistics)
target_include_directories(arbitrium_decision PUBLIC include "${CMAKE_BINARY_DIR}")
target_compile_definitions(arbitrium_decision PUBLIC ARBITRIUM_BUILD_DIGEST="${ARBITRIUM_BUILD_DIGEST}")
add_executable(arbitrium_driver src/driver.cpp)
target_link_libraries(arbitrium_driver PRIVATE arbitrium_decision)
enable_testing()
add_test(NAME decision_binding COMMAND arbitrium_driver binding-self-test)
